@GetMapping("/appVersion.do")
public String appVersion(Model model) {
    String version = "";
    try (InputStream ips = getClass()
            .getResourceAsStream("/META-INF/maven/dcv.card/dcv-btsdebitcard/pom.properties")) {
        if (ips != null) {
            Properties prop = new Properties();
            prop.load(ips);
            version = prop.getProperty("version");
        }
    } catch (Exception e) {
        log.error("Error loading version", e);
    }
    model.addAttribute("appVersion", version);
    return "appVersion"; // JSP or Thymeleaf page name
}


<h1>Running app version is: ${appVersion}</h1>
