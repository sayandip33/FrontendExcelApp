let isPosting = false;
    const btn = document.getElementById("postBtn");

    btn.addEventListener("click", function () {
      if (isPosting) return;

      isPosting = true;
      btn.disabled = true;

      console.log("POST data:", {
        message: "Hello from frontend!",
        time: new Date().toISOString()
      });

      setTimeout(() => {
        isPosting = false;
        btn.disabled = false;
      }, 2000); // simulate delay
    });
